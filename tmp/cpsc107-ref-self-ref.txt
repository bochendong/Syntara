Reference

某些数据定义包含已定义的其他数据定义

常见情况: 复合数据(compound data)包含其他自定义数据。

(@htdd Ball)
(define-struct ball (x y))
;; Ball is (make-ball Number Number)
;; interp. a ball at position x, y
(define BALL-1 (make-ball 6 10))


(@dd-template-rules compound)        ;2 fields
(define (fn-for-ball b)
  (... (ball-x b)        ;Number
      (ball-y b)))       ;Number


(@htdd Game)
(define-struct game (ball score))
;; Game is (make-game Ball Number)
;; interp. the current state of a game, with the ball and score


(define GAME-1 (make-game (make-ball 1 5) 2))


(@dd-template-rules compound               ; 2 fields
                           ref)            ; Ball


(define (fn-for-game g)
  (... (fn-for-ball (game-ball g))
      (game-score g)))                ;Number

(@htdd Dot)
(define-struct dot (x y))
;; Dot is (make-dot Integer Integer)
;; interp. A dot on the screen, w/ x and y coordinates in pixels.
(define D1 (make-dot 10 30))


(@dd-template-rules compound) ;2 fields


(define (fn-for-dot d)
  (... (dot-x d)     ;Integer
        (dot-y d))) ;Integer


(@htdd ListOfDot)
;; ListOfDot is one of:
;; - empty
;; - (cons Dot ListOfDot)
;; interp. a list of Dot
(define LOD1 empty)
(define LOD2 (cons (make-dot 10 20) (cons (make-dot 3 6) empty)))


(@dd-template-rules one-of                          ;2 cases
                           atomic-distinct        ;empty
                           compound                  ;(cons Dot ListOfDot)
                           ref                    ;(first lod) is Dot
                           self-ref)             ;(rest lod) is ListOfDot


(define (fn-for-lod lod)
  (cond [(empty? lod) (...)]
         [else
           (... (fn-for-dot (first lod))
                   (fn-for-lod (rest lod)))]))

(@dd-template-rules one-of                   ;4 subclasses

                           atomic-distinct ;"fee"

                           atomic-distinct ;"floo"

                           compound            ;(make-fie Integer String)

                           compound            ;(make-foe Number Waldo)

                           self-ref        ;(foe-d w) is Waldo

                           atomic-distinct);true



(define (fn-for-waldo w)

  (cond [(and (string? w) (string=? w "fee")) (...)]

        [(string? w) (...)]

        [(fie? w)

          (... (fie-a w)

                (fie-b w))]

        [(foe? w)

          (... (foe-c w)

                (fn-for-waldo (foe-d w)))]

        [else (...)]))
